import (

	_ "github.com/go-sql-driver/mysql"

func TestM4AFormalCarrierFixture(t *testing.T) {
	if os.Getenv("QS_M4_CARRIER_FIXTURE_ACK") != "owned_synthetic_only" {
		t.Fatal("fixed owned fixture acknowledgment required")
	}
	mode := os.Getenv("QS_M4_CARRIER_FIXTURE_ACTION")
	if mode != "seed" && mode != "partial" && mode != "target" && mode != "witness" {
		t.Fatal("fixed fixture action required")
	}
	db, err := sql.Open("mysql", "root@tcp(mysql:3306)/qs?parseTime=true&multiStatements=true&charset=utf8")
	if err != nil {
		t.Fatal(err)
	}
	db.SetMaxOpenConns(1)
	client, err := mongo.Connect(t.Context(), options.Client().ApplyURI("mongodb://mongo:27017/?replicaSet=rs0&directConnection=true"))
	if err != nil {
		_ = db.Close()
		t.Fatal(err)
	}
	defer db.Close()
	defer client.Disconnect(context.Background())
	if mode == "seed" {
		resources := migrationResourcesThrough(t, 100, 39)
		if resources.sha256 != "1af415423e4382d35108a400d1c93913bb17abb277b860560661710447b8a95b" {
			t.Fatal("original resource drift")
		}
		cfg, approval := testBootstrapApproval(t, resources)
		cfg.MySQLDatabase, cfg.MongoDatabase, cfg.ExpectedSourceSHA = "qs", "qs", "51163edab7038adcdb6b8f1712d6dc434f361dc6"
		approval.MySQLDatabase, approval.MongoDatabase, approval.ApprovedSourceSHA = cfg.MySQLDatabase, cfg.MongoDatabase, cfg.ExpectedSourceSHA
		approval.MySQLIdentitySHA256, approval.MongoClusterSHA256, err = pairIdentity(t.Context(), db, client, "qs", "qs")
		if err != nil {
			t.Fatal(err)
		}
		writeBootstrapApproval(t, &cfg, approval)
		pair := nativeSchemaPair{db: db, client: client, mongoDB: client.Database("qs"), cfg: cfg}
		pair.runBoth(t, pair.preflight(t, resources))
		seedHistoricalEvidenceRows(t, pair)
		for done := 0; done < 247461; done += 1000 {
			n := 1000
			if left := 247461 - done; left < n {
				n = left
			}
			var a, o strings.Builder
			a.WriteString("INSERT INTO assessment(id,org_id,testee_id,questionnaire_code,questionnaire_version,answer_sheet_id,origin_type,status,evaluation_model_kind,evaluation_model_code,evaluation_model_version,primary_score_value) VALUES ")
			o.WriteString("INSERT INTO evaluation_outcome(id,org_id,assessment_id,testee_id,evaluation_run_id,model_kind,model_code,model_version,payload_json,report_input_json,evaluated_at,committed_event_id,committed_event_evidence) VALUES ")
			for i := 0; i < n; i++ {
				if i > 0 {
					a.WriteByte(',')
					o.WriteByte(',')
				}
				id := 10000000 + done + i
				fmt.Fprintf(&a, "(%d,1,2,'thin-owned','1',%d,'direct','evaluated','scale','thin-owned','1',11.5)", id, id)
				fmt.Fprintf(&o, "(%d,1,%d,2,'thin-%d','scale','thin-owned','1',CONCAT('{\"fixture\":\"',REPEAT('x',6130),'\"}'),'{\"fixture\":\"thin retained\"}','2026-10-13 00:00:00','thin-event-%d','{\"version\":1}')", id+10000000, id, id, id)
			}
			for _, query := range []string{a.String(), o.String()} {
				ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
				_, e := db.ExecContext(ctx, query)
				cancel()
				if e != nil {
					t.Fatal("fixed capacity insert failed", e)
				}
			}
		}
		if err = pair.preflight(t, resources).VerifyComplete(t.Context()); err != nil {
			t.Fatal(err)
		}
	} else if mode == "partial" || mode == "target" {
		// This is the existing native matrix's fixture-only partial-DDL setup.
		// No production force/down/dirty clearing command is available.
		var version uint
		var dirty bool
		if db.QueryRowContext(t.Context(), "SELECT version,dirty FROM schema_migrations").Scan(&version, &dirty) != nil || version != 100 || dirty {
			t.Fatal("clean original100 required")
		}
		if _, err = db.ExecContext(t.Context(), "UPDATE schema_migrations SET version=101,dirty=true"); err != nil {
			t.Fatal(err)
		}
		if _, err = db.ExecContext(t.Context(), "ALTER TABLE assessment DROP CHECK chk_assessment_historical_lifecycle, DROP COLUMN historical_lifecycle_evidence"); err != nil {
			t.Fatal(err)
		}
		if mode == "target" {
			if _, err = db.ExecContext(t.Context(), "ALTER TABLE evaluation_outcome DROP CHECK chk_outcome_historical_committed, DROP COLUMN historical_committed_evidence"); err != nil {
				t.Fatal(err)
			}
			if _, err = db.ExecContext(t.Context(), "UPDATE schema_migrations SET dirty=false WHERE version=101 AND dirty=true"); err != nil {
				t.Fatal(err)
			}
		}
	}
	// Independent counts/NULL, complete ledger, exact original SHOW CREATE and
	// native dual identity are witnesses; the Tool additionally compares every
	// retained row and the complete Mongo raw ledger/UUID against its capture.
	result := map[string]any{"action": mode, "production": false, "business_runtime_started": false}
	var version uint
	var dirty bool
	if err = db.QueryRowContext(t.Context(), "SELECT version,dirty FROM schema_migrations").Scan(&version, &dirty); err != nil {
		t.Fatal(err)
	}
	result["mysql_version"], result["mysql_dirty"] = version, dirty
	pair := nativeSchemaPair{db: db, client: client, mongoDB: client.Database("qs")}
	_, _, mongoVersion, mongoDirty := pair.heads(t)
	if mongoVersion != 39 || mongoDirty {
		t.Fatal("original Mongo clean39 changed")
	}
	result["mongo_version"], result["mongo_dirty"] = mongoVersion, mongoDirty
	for _, table := range []string{"assessment", "evaluation_outcome"} {
		var n int
		if db.QueryRowContext(t.Context(), "SELECT COUNT(*) FROM "+table).Scan(&n) != nil || n != 247463 {
			t.Fatal("full count drift")
		}
		result[table+"_count"] = n
		var name, ddl string
		if db.QueryRowContext(t.Context(), "SHOW CREATE TABLE "+table).Scan(&name, &ddl) != nil {
			t.Fatal("independent schema unavailable")
		}
		result[table+"_ddl"] = ddl
		column := "historical_lifecycle_evidence"
		if table == "evaluation_outcome" {
			column = "historical_committed_evidence"
		}
		if strings.Contains(ddl, "`"+column+"`") {
			var nonNull int
			if db.QueryRowContext(t.Context(), "SELECT COUNT(*) FROM "+table+" WHERE "+column+" IS NOT NULL").Scan(&nonNull) != nil || nonNull != 0 {
				t.Fatal("historical values changed")
			}
		}
	}
	result["mysql_identity_sha256"], result["mongo_identity_sha256"], err = pairIdentity(t.Context(), db, client, "qs", "qs")
	if err != nil {
		t.Fatal(err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err = client.Disconnect(ctx); err != nil {
		t.Fatal(err)
	}
	result["sql_pool_closed"], result["mongo_closed"], result["complete"] = true, true, true
	raw, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	fmt.Println("A_FORMAL_CARRIER_FIXTURE " + string(raw))
}
